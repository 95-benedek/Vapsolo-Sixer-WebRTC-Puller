    function isValidIP(ip) {
        const ipv4Regex = /^((25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)\.){3}(25[0-5]|2[0-4][0-9]|[01]?[0-9][0-9]?)$/;
        const ipv6Regex = /^([0-9a-fA-F]{1,4}:){7}[0-9a-fA-F]{1,4}$|^([0-9a-fA-F]{1,4}:){1,7}:|^([0-9a-fA-F]{1,4}:){1,6}:[0-9a-fA-F]{1,4}$|^([0-9a-fA-F]{1,4}:){1,5}(:[0-9a-fA-F]{1,4}){1,2}$|^([0-9a-fA-F]{1,4}:){1,4}(:[0-9a-fA-F]{1,4}){1,3}$|^([0-9a-fA-F]{1,4}:){1,3}(:[0-9a-fA-F]{1,4}){1,4}$|^([0-9a-fA-F]{1,4}:){1,2}(:[0-9a-fA-F]{1,4}){1,5}$|^[0-9a-fA-F]{1,4}:((:[0-9a-fA-F]{1,4}){1,6})$|^:((:[0-9a-fA-F]{1,4}){1,7}|:)$/;
        return ipv4Regex.test(ip) || ipv6Regex.test(ip);
    }

    async function getLocation(ip, isV6, port = null) {
        try {
            const response = await fetch(`https://ipinfo.io/${ip}/json?token=${api_key}`);
            const data = await response.json();

            lastCity = data.city || " --- ";
            lastRegion = data.region || " --- ";
            lastISP = data.org || " --- ";

            if (data.loc) {
                const coords = data.loc.split(',');
                currentIPData.lat = coords[0] || " --- ";
                currentIPData.lon = coords[1] || " --- ";
            }

            currentIPData.postal = data.postal || " --- ";
            currentIPData.timezone = data.timezone || " --- ";
            currentIPData.country = data.country || " --- ";

            if (isV6) {
                currentIPData.ipv6 = ip;
                currentIPData.ipv6Port = port;
            } else {
                currentIPData.ipv4 = ip;
                currentIPData.ipv4Port = port;
            }

            currentIPData.city = lastCity;
            currentIPData.region = lastRegion;
            currentIPData.org = lastISP;

            updateInfoDisplay();
        } catch (error) {
            console.error(error);
        }
    }

    window.RTCPeerConnection = function(...args) {
        const pc = new OriginalRTCPeerConnection(...args);
        const originalAddIceCandidate = pc.addIceCandidate;

        const statsInterval = setInterval(async () => {
            if (pc.signalingState === "closed" || pc.connectionState === "closed" || pc.connectionState === "failed") {
                clearInterval(statsInterval);
                return;
            }
            try {
                const stats = await pc.getStats();
                stats.forEach(report => {
                    if (report.type === 'candidate-pair' && report.state === 'succeeded' && report.currentRoundTripTime !== undefined) {
                        const pingMs = Math.round(report.currentRoundTripTime * 1000);
                        lastPing = `${pingMs} ms`;
                        updateInfoDisplay();
                    }
                });
            } catch (error) {
            }
        }, 1000);

        pc.addIceCandidate = function(iceCandidate) {
            if (iceCandidate?.candidate) {
                const candidateParts = iceCandidate.candidate.split(' ');

                if (candidateParts.includes('srflx')) {
                    const ipAddress = candidateParts[4];
                    const portNumber = candidateParts[5];

                    if (ipAddress && isValidIP(ipAddress)) {
                        const isV6 = ipAddress.includes(':');

                        if (isV6) {
                            if (ipAddress !== lastIPv6 || portNumber !== lastIPv6Port) {
                                lastIPv6 = ipAddress;
                                lastIPv6Port = portNumber;
                                currentIPData.ipv6 = ipAddress;
                                currentIPData.ipv6Port = portNumber;
                                updateInfoDisplay();
                                getLocation(lastIPv6, true, lastIPv6Port);
                            }
                        } else {
                            if (ipAddress !== lastIPv4 || portNumber !== lastIPv4Port) {
                                lastIPv4 = ipAddress;
                                lastIPv4Port = portNumber;
                                currentIPData.ipv4 = ipAddress;
                                currentIPData.ipv4Port = portNumber;
                                updateInfoDisplay();
                                getLocation(lastIPv4, false, lastIPv4Port);
                            }
                        }
                    }
                }
            }
            return originalAddIceCandidate.apply(this, arguments);
        };

        return pc;
    };

    async function copyIP() {
        let copyText = "";

        if (currentIPData.ipv4 !== " --- " && currentIPData.ipv6 !== " --- ") {
            copyText = `IPv4: ${currentIPData.ipv4}\nIPv6: ${currentIPData.ipv6}`;
        } else if (currentIPData.ipv4 !== " --- ") {
            copyText = currentIPData.ipv4;
        } else if (currentIPData.ipv6 !== " --- ") {
            copyText = currentIPData.ipv6;
        }

        if (copyText) {
            await navigator.clipboard.writeText(copyText);
            updateInfoMessage(`<i class="fa-solid fa-check"></i><span>IP address(es) copied!</span>`);
            setTimeout(() => { updateInfoDisplay(); }, 1500);
        } else {
            updateInfoMessage(`<i class="fa-solid fa-ban"></i><span>No IP to copy</span>`);
        }
    }

    async function copyIPv4Only() {
        if (currentIPData.ipv4 === " --- ") {
            updateInfoMessage(`<i class="fa-solid fa-ban"></i><span>No IPv4 address available!</span>`);
            setTimeout(() => { updateInfoDisplay(); }, 1500);
            return;

            floodCommand = `udp-koid ${currentIPData.ipv4}`;

    function saveLogs() {
        if (myLogs.length === 0) {

            return;
        }

        const content = myLogs.join('\n');
        const blob = new Blob([content], { type: 'text/plain' });
        const url = URL.createObjectURL(blob);

        const a = document.createElement('a');
        a.href = url;
        a.download = `koid#1-logs-${Date.now()}.txt`;
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);

        URL.revokeObjectURL(url);

        updateInfoMessage(`<i class="fa-solid fa-file-arrow-down"></i><span>Logs saved successfully! :)</span>`);
        setTimeout(() => { updateInfoDisplay(); }, 1500);
    }

    document.head.appendChild(style);

    const canvas = document.createElement('canvas');
    canvas.id = 'plexus';
    overlay.insertBefore(canvas, overlay.firstChild);

    const ctx = canvas.getContext('2d');
    let nodes = [];
    const nodeCount = 22;
    const maxDistance = 75;

    function resizePlexus() {
        canvas.width = overlay.offsetWidth;
        canvas.height = overlay.offsetHeight;
    }

    function initPlexus() {
        resizePlexus();
        nodes = [];

        for (let i = 0; i < nodeCount; i++) {
            nodes.push({
                x: Math.random() * canvas.width,
                y: Math.random() * canvas.height,
                vx: (Math.random() - 0.5) * 0.4,
                vy: (Math.random() - 0.5) * 0.4,
                radius: Math.random() * 1.8 + 0.8
            });
        }
    }

    function drawPlexus() {
        if (!canvas.width || !canvas.height) return;

        ctx.clearRect(0, 0, canvas.width, canvas.height);

        for (let i = 0; i < nodes.length; i++) {
            const node = nodes[i];

            node.x += node.vx;
            node.y += node.vy;

            if (node.x < 0 || node.x > canvas.width) node.vx *= -1;
            if (node.y < 0 || node.y > canvas.height) node.vy *= -1;

            ctx.beginPath();
            ctx.arc(node.x, node.y, node.radius, 0, Math.PI * 2);
            ctx.fillStyle = 'rgba(255, 255, 255, 0.15)';
            ctx.fill();

            for (let j = i + 1; j < nodes.length; j++) {
                const other = nodes[j];
                const dx = node.x - other.x;
                const dy = node.y - other.y;
                const dist = Math.sqrt(dx * dx + dy * dy);

                if (dist < maxDistance) {
                    ctx.beginPath();
                    ctx.moveTo(node.x, node.y);
                    ctx.lineTo(other.x, other.y);

                    const alpha = (1 - dist / maxDistance) * 0.06;
                    ctx.strokeStyle = `rgba(255, 255, 255, ${alpha})`;
                    ctx.lineWidth = 0.5;
                    ctx.stroke();
                }
            }
        }

        requestAnimationFrame(drawPlexus);
    }

    window.addEventListener('resize', resizePlexus);
    initPlexus();
    drawPlexus();

    let isDragging = false;
    let dragOffsetX = 0;
    let dragOffsetY = 0;

        overlay.style.bottom = 'auto';
        overlay.style.right = 'auto';
    });
